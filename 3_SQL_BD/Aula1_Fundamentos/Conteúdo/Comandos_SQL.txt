 7. Filtros com operadores lógicos - usados no comando WHERE
	1. > maior que 
	2. < menor que 
	3. >= maior ou igual que 
		select * from tabelaclientes where nome_cliente > 'C'
		select * from tabelapedidos where data_do_pedido > '2023-09-19';
	4. < menor ou igual que 
	5. = igual que 
		select * from tabelapedidos where total_do_pedido = 200;
	6. <> Diferente que 
	7. AND - junta duas condições p/ mostrar 1 respota; 
		select * from tabelaprodutos WHERE preco_de_compra >= 500 AND preco_de_compra < 1000;
	8. OR  - mostras duas condições 
		select * from tabelapedidos where status = 'Pendente' OR status = 'Processando';
	9. NOT - Nega uma condição e não a exibe
		select * from tabelapedidos where NOT  status = 'Pendente'
	10. BETWEEN - data entre um intervalo
		select * from tabelapedidos where data_de_envio_es BETWEEN '2023-08-01' AND '2023-09-01'; 
		SELECT * FROM tabelaprodutos WHERE preco_de_compra 
		BETWEEN 200 AND 600 ORDER BY nome_do_produtos;
	11. ORDEM BY - Ordenação por uma coluna
		DESC - do maior para o menor
